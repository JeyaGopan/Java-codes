class Solution {
    public int fib(int n) 
    {
        int frstno=0,scndno=1;
        if(n==0)
        {
        return 0;
        }
        if(n==1)
        {
        return 1;
        }
        for(int i=2;i<=n;i++)
        {
            int thrdno=frstno+scndno;
            frstno=scndno;
            scndno=thrdno;
        }
        return scndno;
    }
}
